cmake_minimum_required(VERSION 3.20)
project(RuntimeIsolation LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The runtime tools require Windows x86 (Win32)")
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
add_library(workspace_profile SHARED workspace_profile.cpp)
target_link_libraries(workspace_profile PRIVATE advapi32)
add_executable(workspace_launcher workspace_launcher.cpp)
target_link_libraries(workspace_launcher PRIVATE advapi32 bcrypt)
add_executable(registry_probe registry_probe.cpp)
target_link_libraries(registry_probe PRIVATE advapi32)
foreach(target workspace_profile workspace_launcher registry_probe)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(${target} PRIVATE /W4 /utf-8)
endforeach()
